POST /predict

          
Classify a single image as high or low eczema severity. multipart/form-data.


          
Parameters

          

            
NameTypeDefaultDescription

            
              
imagefile—JPEG or PNG, max 10 MB. Required.

              
modelstringbothOne of both, cnn, mobilenet.

              
gradcambooltrueWhether to compute the Grad-CAM overlay + heatmap.

            
          


          
Code examples

          

            
            
            
          

          
import requests

with open("skin.jpg", "rb") as f:
    r = requests.post(
        "http://54.226.38.244:8000/api/v1/predict",
        files={"image": f},
        data={"model": "both", "gradcam": "true"},
    )
print(r.json())

          
curl -X POST http://54.226.38.244:8000/api/v1/predict \
  -F "image=@skin.jpg" \
  -F "model=both" \
  -F "gradcam=true"

          
const form = new FormData();

POST /predict/compare

          
Compare two images and report which is more severe, and by how much.


          
Parameters

          

            
NameTypeDefaultDescription

            
              
image_afile—First image (JPEG/PNG). Required.

              
image_bfile—Second image (JPEG/PNG). Required.

              
modelstringbothOne of both, cnn, mobilenet.

            
          


          
Code examples

          

            
            
            
          

          
import requests

files = {
    "image_a": open("before.jpg", "rb"),
    "image_b": open("after.jpg", "rb"),
}
r = requests.post(
    "http://54.226.38.244:8000/api/v1/predict/compare",
    files=files, data={"model": "both"},
)
print(r.json())

          
curl -X POST http://54.226.38.244:8000/api/v1/predict/compare \
  -F "image_a=@before.jpg" \
  -F "image_b=@after.jpg" \
  -F "model=both"

          
const form = new FormData();
form.append("image_a", a.files[0]);
form.append("image_b", b.files[0]);
form.append("model", "both");

const res = await fetch("http://54.226.38.244:8000/api/v1/predict/compare", {
  method: "POST", body: form,
});
console.log(await res.json());
